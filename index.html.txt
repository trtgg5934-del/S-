  <script>
    // إنشاء قلوب متساقطة بالخلفية
    function createFallingHearts() {
      const container = document.getElementById('heartsContainer');
      const heartSymbols = ['❤️', '💖', '💕', '💗', '🌸'];
      for (let i = 0; i < 25; i++) {
        const heart = document.createElement('div');
        heart.className = 'falling-heart';
        heart.innerText = heartSymbols[Math.floor(Math.random() * heartSymbols.length)];
        heart.style.left = Math.random() * 100 + 'vw';
        heart.style.animationDuration = (Math.random() * 3 + 3) + 's';
        heart.style.animationDelay = Math.random() * 5 + 's';
        heart.style.fontSize = (Math.random() * 15 + 15) + 'px';
        container.appendChild(heart);
      }
    }
    createFallingHearts();

    // التحقق من كلمة السر
    function checkPassword() {
      const input = document.getElementById('passInput').value.trim();
      if (input === "25 اغسطس" || input === "25 أغسطس" || input === "25اغسطس" || input === "25أغسطس") {
        document.getElementById('lockCard').classList.add('hidden');
        document.getElementById('mainCard').classList.remove('hidden');
        celebrate();
      } else {
        document.getElementById('errorMsg').classList.remove('hidden');
      }
    }

    function goToReasonsCard() {
      document.getElementById('mainCard').classList.add('hidden');
      document.getElementById('reasonsCard').classList.remove('hidden');
      nextReason();
    }

    // أسباب حب شهد
    const reasons = [
      "ضحكتج اللي تفتح النفس وتغير لي المود 🥰",
      "لأنج ألطف وأحلى شيء صار لي بحياتي ✨",
      "تفاصيلج الصغيرة والجميلة اللي تميزج عن الكل 🌸",
      "لأنج دايماً تخليني أحس إني محظوظ فيج ❤️",
      "لأنج شهد.. وأجمل من شهد ما فيه! 💖"
    ];
    let reasonIndex = 0;

    function nextReason() {
      const box = document.getElementById('reasonText');
      box.innerText = reasons[reasonIndex];
      reasonIndex = (reasonIndex + 1) % reasons.length;
    }

    function goToLoveQuestion() {
      document.getElementById('reasonsCard').classList.add('hidden');
      document.getElementById('questionCard').classList.remove('hidden');
    }

    function sayYes() {
      celebrate();
      document.getElementById('questionCard').classList.add('hidden');
      document.getElementById('loveMoreCard').classList.remove('hidden');
    }

    function goToKittyGame() {
      document.getElementById('loveMoreCard').classList.add('hidden');
      document.getElementById('gameCard').classList.remove('hidden');
    }

    // لعبة هيلو كيتي
    let step = 0;
    function bringKittiesCloser() {
      step++;
      const kLeft = document.getElementById('kittyLeft');
      const kRight = document.getElementById('kittyRight');
      const heart = document.getElementById('heartCenter');
      const btn = document.getElementById('bringCloserBtn');

      if (step === 1) {
        kLeft.style.transform = "translateX(-30px)";
        kRight.style.transform = "translateX(30px)";
        heart.innerText = "💗";
        heart.style.opacity = "0.7";
      } else if (step === 2) {
        kLeft.style.transform = "translateX(-60px)";
        kRight.style.transform = "translateX(60px)";
        heart.innerText = "💖";
        heart.style.opacity = "1";
        heart.style.transform = "scale(1.3)";
        btn.innerText = "كمان شوية! ✨";
      } else if (step === 3) {
        kLeft.style.transform = "translateX(-85px)";
        kRight.style.transform = "translateX(85px)";
        heart.innerText = "❤️";
        heart.style.transform = "scale(1.8)";
        btn.innerText = "اكتمل الحب! 🎉";
        celebrate();
        btn.disabled = true;
        btn.style.opacity = "0.7";
      }
    }

    function celebrate() {
      confetti({
        particleCount: 100,
        spread: 70,
        origin: { y: 0.6 }
      });
    }
